# Generators are discovered: every generators/<owner>/<plugin>/<version>/CMakeLists.txt calls
# uni_generator(LIBRARIES <libraries it uses>). The version directory is a unity build: only its
# generator.c is compiled (it includes the other .c files of the directory) and it exports
# `const UniGenerator gen_<owner>_<plugin>_<version>` (the path as a C identifier). The registry
# (registry.c in the build directory) lists the selected generators sorted by name and version.
set(UNI_GENERATORS "all" CACHE STRING
  "generators to build: all, or <owner>/<plugin>[/<version>] entries separated by , or ;")

add_library(uni_generators STATIC ${CMAKE_CURRENT_BINARY_DIR}/registry.c)
target_include_directories(uni_generators PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(uni_generators PUBLIC uni_uni)
set_property(GLOBAL PROPERTY UNI_GENERATOR_KEYS "")

function(uni_generator)
  cmake_parse_arguments(G "" "" "LIBRARIES" ${ARGN})
  file(RELATIVE_PATH key ${PROJECT_SOURCE_DIR}/generators ${CMAKE_CURRENT_SOURCE_DIR})
  target_sources(uni_generators PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/generator.c)
  target_link_libraries(uni_generators PUBLIC ${G_LIBRARIES})
  set_property(GLOBAL APPEND PROPERTY UNI_GENERATOR_KEYS ${key})
endfunction()

string(REPLACE "," ";" _uni_selected "${UNI_GENERATORS}")
file(GLOB _uni_dirs LIST_DIRECTORIES true ${CMAKE_CURRENT_SOURCE_DIR}/*/*/v*)
foreach(dir IN LISTS _uni_dirs)
  if(NOT EXISTS ${dir}/CMakeLists.txt)
    continue()
  endif()
  file(RELATIVE_PATH key ${CMAKE_CURRENT_SOURCE_DIR} ${dir})
  set(selected FALSE)
  foreach(entry IN LISTS _uni_selected)
    string(STRIP "${entry}" entry)
    string(LENGTH "${entry}/" n)
    string(SUBSTRING "${key}/" 0 ${n} prefix)
    if(entry STREQUAL "all" OR prefix STREQUAL "${entry}/")
      set(selected TRUE)
    endif()
  endforeach()
  if(selected)
    add_subdirectory(${dir} ${CMAKE_CURRENT_BINARY_DIR}/${key})
  endif()
endforeach()

# registry.c: "<name> <version>" sort keys (like `sort -k1,1 -k2,2V`), then the symbols.
get_property(_uni_keys GLOBAL PROPERTY UNI_GENERATOR_KEYS)
set(_uni_sort)
foreach(key IN LISTS _uni_keys)
  string(REGEX REPLACE "^(.*)/([^/]*)$" "\\1 \\2" s ${key})
  list(APPEND _uni_sort "${s}")
endforeach()
list(SORT _uni_sort COMPARE NATURAL)
set(_uni_externs "")
set(_uni_entries "")
foreach(s IN LISTS _uni_sort)
  string(REPLACE " " "/" key "${s}")
  string(MAKE_C_IDENTIFIER "gen_${key}" sym)
  string(APPEND _uni_externs "extern const UniGenerator ${sym};\n")
  string(APPEND _uni_entries "\t&${sym},\n")
endforeach()
if(_uni_entries STREQUAL "")
  message(FATAL_ERROR "UNI_GENERATORS=${UNI_GENERATORS} selects no generator")
endif()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/registry.c CONTENT "/* Generated by generators/CMakeLists.txt. */
#include \"registry.h\"

${_uni_externs}
const UniGenerator *const uni_generators[] = {
${_uni_entries}};

const int uni_n_generators = (int)(sizeof uni_generators / sizeof *uni_generators);
")
