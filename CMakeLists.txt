# Unit tests: one executable per test file (tests/unit/test.h), run by ctest.
# Built only for the host or under an emulator; uses the same flags as the binary.
# With a subset of UNI_GENERATORS, uni_register_unit_tests() at the end keeps only the tests of
# libraries the selected generators link: the others are not built and are disabled.
function(uni_add_unit_test name)
  cmake_parse_arguments(T "" "" "SOURCES;LIBRARIES;INCLUDES" ${ARGN})
  add_executable(${name} ${T_SOURCES})
  target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/tests/unit ${T_INCLUDES})
  target_link_libraries(${name} PRIVATE ${T_LIBRARIES})
  add_test(NAME unit/${name} COMMAND ${name})
  set_tests_properties(unit/${name} PROPERTIES LABELS unit)
  set_property(GLOBAL APPEND PROPERTY UNI_UNIT_TESTS ${name})
endfunction()

# Targets linked by `target`, transitively (link items that are targets only).
function(_uni_link_closure target out)
  set(seen)
  set(todo ${target})
  while(todo)
    list(POP_FRONT todo t)
    if(NOT TARGET ${t} OR t IN_LIST seen)
      continue()
    endif()
    list(APPEND seen ${t})
    get_target_property(libs ${t} LINK_LIBRARIES)
    get_target_property(ilibs ${t} INTERFACE_LINK_LIBRARIES)
    foreach(l IN LISTS libs ilibs)
      if(l AND NOT l MATCHES "NOTFOUND$")
        list(APPEND todo ${l})
      endif()
    endforeach()
  endwhile()
  set(${out} ${seen} PARENT_SCOPE)
endfunction()

function(uni_register_unit_tests)
  if(UNI_GENERATORS STREQUAL "all")
    return()
  endif()
  get_property(tests GLOBAL PROPERTY UNI_UNIT_TESTS)
  _uni_link_closure(protoc-gen-uni used)
  # Unit tests belong to the directory that defines them; it is switched off as a whole (also the
  # extra add_test runs of the same executables) when none of the libraries it defines is used.
  set(dirs)
  foreach(name IN LISTS tests)
    get_target_property(dir ${name} SOURCE_DIR)
    list(APPEND dirs ${dir})
  endforeach()
  list(REMOVE_DUPLICATES dirs)
  foreach(dir IN LISTS dirs)
    get_property(targets DIRECTORY ${dir} PROPERTY BUILDSYSTEM_TARGETS)
    set(libs)
    set(exes)
    foreach(t IN LISTS targets)
      get_target_property(type ${t} TYPE)
      if(type MATCHES "LIBRARY$")
        list(APPEND libs ${t})
      elseif(t IN_LIST tests)
        list(APPEND exes ${t})
      endif()
    endforeach()
    set(wanted TRUE)
    if(libs)
      set(wanted FALSE)
      foreach(l IN LISTS libs)
        if(l IN_LIST used)
          set(wanted TRUE)
        endif()
      endforeach()
    endif()
    if(NOT wanted)
      set_property(TARGET ${exes} PROPERTY EXCLUDE_FROM_ALL TRUE)
      get_property(dir_tests DIRECTORY ${dir} PROPERTY TESTS)
      foreach(t IN LISTS dir_tests)
        set_property(TEST ${t} DIRECTORY ${dir} PROPERTY DISABLED TRUE)
      endforeach()
    endif()
  endforeach()
endfunction()

# Libraries: every top-most CMakeLists.txt under pkg/ (pkg/core: util.h, sys.h; pkg/pb: wire
# format, schemas; pkg/uni: generator selection; pkg/languages/<language>/...: shared by the
# generators of one language; ...), then generators/ (discovered generators and the registry).
# Build a single target (e.g. --target protoc-gen-uni) to compile only what it needs.
file(GLOB_RECURSE _uni_pkg_lists ${CMAKE_SOURCE_DIR}/pkg/*/CMakeLists.txt)
list(SORT _uni_pkg_lists)
set(_uni_pkg_dirs)
foreach(f IN LISTS _uni_pkg_lists)
  get_filename_component(dir ${f} DIRECTORY)
  set(nested FALSE)
  foreach(d IN LISTS _uni_pkg_dirs)
    string(FIND "${dir}/" "${d}/" pos)
    if(pos EQUAL 0)
      set(nested TRUE)
    endif()
  endforeach()
  if(NOT nested)
    list(APPEND _uni_pkg_dirs ${dir})
    add_subdirectory(${dir})
  endif()
endforeach()
